# C FFI — Phase 1 Implementation Plan (ABI core + primitives)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A compiled March program can bind an existing C symbol by explicit name and call it with `Int`/`Float`/`Bool` arguments and return values, on a documented, versioned ABI foundation (`runtime/march_ffi.h` / `march_ffi.c`).

**Architecture:** Extend the existing `extern` scaffold (parser→AST→typecheck→TIR→codegen). Add a stable C header with value accessors + RC ops (the `erl_nif.h` analog), wire its `.c` into the runtime build, add `= "symbol"` syntax so existing C symbols can be bound directly, and ensure the codegen boundary unwraps/rewraps primitives by static type (Int `ashr`, Bool `ashr`+trunc, Float `bitcast`, heap verbatim).

**Tech Stack:** OCaml 5.3 (compiler), menhir (parser), C11 (runtime), dune + alcotest (tests). Spec: `specs/2026-06-19-c-ffi-abi-design.md`.

**Representation facts (verified, do not re-derive):**
- Value word: tagged `Int` = `(n<<1)|1`, untag = `v>>1` (arithmetic). `Bool` same (0/1).
- `Float` = `bitcast double→i64` (untagged, raw IEEE-754 bits). Access by static type only.
- Heap pointer = even, non-zero word, used verbatim (no untag, never `|1`).
- Header: `march_hdr { int64_t rc; int32_t tag; int32_t pad; }` (16 bytes). `march_alloc`/`march_incrc`/`march_decrc` in `runtime/march_runtime.h`.
- Runtime `.c` files are listed in `bin/main.ml` (~line 266–313) and `test/test_helpers.ml` (~line 510–513); every `.h` in `runtime/` is auto-digested into the cache key, so adding files invalidates caches automatically.

### Task 1: ABI header `runtime/march_ffi.h`

**Files:**
- Create: `runtime/march_ffi.h`

- [ ] **Step 1: Write the header** (signatures only; impl in Task 2)

```c
/* runtime/march_ffi.h — March External Function Interface, stable ABI.
 * Bindings include ONLY this header; runtime internals stay private.
 * See specs/2026-06-19-c-ffi-abi-design.md §4. */
#ifndef MARCH_FFI_H
#define MARCH_FFI_H
#include <stdint.h>
#include <stddef.h>

#define MARCH_FFI_ABI_VERSION 1
int32_t march_ffi_abi_version(void);   /* returns MARCH_FFI_ABI_VERSION */

typedef int64_t march_value;           /* opaque tagged value word (§4.3) */

/* Primitive accessors. Access by the value's STATIC type — there is no
 * universal runtime discriminator (a Float word may be even or odd). */
march_value march_make_int(int64_t n);   /* (n<<1)|1 */
int64_t     march_get_int(march_value v);/* v>>1 (arithmetic) */
march_value march_make_bool(int b);
int         march_get_bool(march_value v);
march_value march_make_float(double f);  /* bitcast, untagged */
double      march_get_float(march_value v);

int    march_is_heap(march_value v);     /* even && non-zero */
void  *march_as_ptr(march_value v);      /* heap pointer, verbatim */
march_value march_from_ptr(void *p);     /* wrap pointer, verbatim */

/* Reference counting — tag-aware. ONLY valid on Int/Bool/heap value words,
 * NEVER on a Float word (its bits may look like a heap pointer). */
march_value march_dup(march_value v);    /* heap: incrc; tagged: no-op; returns v */
void        march_drop(march_value v);   /* heap: decrc (may free); tagged: no-op */

#endif /* MARCH_FFI_H */
```

```c
#include "../runtime/march_ffi.h"
#include <assert.h>
#include <stdio.h>
#include <string.h>

int main(void) {
    /* version handshake */
    assert(march_ffi_abi_version() == MARCH_FFI_ABI_VERSION);

    /* int round-trip + tag layout (1 -> 3, 2 -> 5, matches codegen) */
    assert(march_make_int(1) == 3);
    assert(march_make_int(2) == 5);
    assert(march_get_int(march_make_int(42)) == 42);
    assert(march_get_int(march_make_int(-7)) == -7);

    /* bool */
    assert(march_get_bool(march_make_bool(1)) == 1);
    assert(march_get_bool(march_make_bool(0)) == 0);

    /* float bitcast round-trip */
    double f = 3.14159;
    assert(march_get_float(march_make_float(f)) == f);

    /* heap tag detection: tagged ints are not heap; even non-zero is heap */
    assert(!march_is_heap(march_make_int(5)));
    int dummy; assert(march_is_heap(march_from_ptr(&dummy)));
    assert(!march_is_heap(march_from_ptr(NULL)));

    /* dup/drop are no-ops on tagged ints (must not crash / change value) */
    assert(march_dup(march_make_int(9)) == march_make_int(9));
    march_drop(march_make_int(9));

    printf("test_ffi: OK\n");
    return 0;
}
```

```c
/* runtime/march_ffi.c — see march_ffi.h */
#include "march_ffi.h"
#include "march_runtime.h"   /* march_incrc / march_decrc */
#include <string.h>

int32_t march_ffi_abi_version(void) { return MARCH_FFI_ABI_VERSION; }

march_value march_make_int(int64_t n)   { return (n << 1) | 1; }
int64_t     march_get_int(march_value v){ return v >> 1; }
march_value march_make_bool(int b)      { return ((int64_t)(b ? 1 : 0) << 1) | 1; }
int         march_get_bool(march_value v){ return (int)(v >> 1) != 0; }

march_value march_make_float(double f)  { int64_t w; memcpy(&w, &f, 8); return w; }
double      march_get_float(march_value v){ double f; memcpy(&f, &v, 8); return f; }

int   march_is_heap(march_value v)  { return v != 0 && (v & 1) == 0; }
void *march_as_ptr(march_value v)   { return (void *)(intptr_t)v; }
march_value march_from_ptr(void *p) { return (march_value)(intptr_t)p; }

march_value march_dup(march_value v)  { if (march_is_heap(v)) march_incrc(march_as_ptr(v)); return v; }
void        march_drop(march_value v) { if (march_is_heap(v)) march_decrc(march_as_ptr(v)); }
```
